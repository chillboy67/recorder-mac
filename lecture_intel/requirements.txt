# ── ASR ──────────────────────────────────────────────────────

# mlx-whisper silently degrades the default `auto` engine to CPU.

faster-whisper>=1.0.0

# ── Speaker separation (diarization) ─────────────────────────

scikit-learn>=1.3.0

# instead, which nothing in the live tree imports.
